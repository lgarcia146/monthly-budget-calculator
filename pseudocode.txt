START

    // This message welcomes the user and explains what the program is for
    DISPLAY "Welcome to the Monthly Budget Calculator"

    // This variable controls whether the program repeats
    SET repeatProgram = "yes"

    // This loop keeps running as long as the user types "yes"
    WHILE repeatProgram == "yes"

        // --- INPUT INCOME ---
        // Ask the user to enter all money received for the month
        // This includes paychecks and any additional income
        PROMPT "Enter your monthly income: "
        GET income

        // This loop checks if the income entered is valid
        // It repeats until the user enters a positive number
        WHILE income is NOT a valid number OR income < 0
            DISPLAY "Invalid entry. Please enter a positive number."
            PROMPT "Enter your monthly income: "
            GET income
        END WHILE


        // --- INPUT EXPENSES ---
        // The next section collects all monthly expenses
        // The user enters estimated amounts for each category

        PROMPT "Enter rent amount: "
        GET rent
        VALIDATE rent      // Makes sure rent is a valid number

        PROMPT "Enter food amount: "
        GET food
        VALIDATE food

        PROMPT "Enter gas amount: "
        GET gas
        VALIDATE gas

        PROMPT "Enter utilities amount: "
        GET utilities
        VALIDATE utilities

        PROMPT "Enter childcare amount: "
        GET childcare
        VALIDATE childcare

        PROMPT "Enter miscellaneous expenses: "
        GET misc
        VALIDATE misc


        // --- CALCULATIONS ---
        // Add all expenses together to get the total monthly spending
        SET totalExpenses = rent + food + gas + utilities + childcare + misc

        // Subtract expenses from income to find out how much money is left
        SET remaining = income - totalExpenses


        // --- DETERMINE STATUS ---
        // This section checks whether the user is over or under budget
        // If remaining money is negative → over budget
        // If remaining money is zero or positive → under budget
        IF remaining < 0 THEN
            SET status = "Over Budget"
        ELSE
            SET status = "Under Budget"
        END IF


        // --- OUTPUT RESULTS ---
        // Display all results so the user can review their monthly budget
        DISPLAY "Total Income: ", income
        DISPLAY "Total Expenses: ", totalExpenses
        DISPLAY "Remaining Balance: ", remaining
        DISPLAY "Status: ", status


        // --- REPEAT OPTION ---
        // Ask the user if they want to calculate another month
        PROMPT "Do you want to calculate another month? (yes/no): "
        GET repeatProgram

        // This loop ensures the user enters only "yes" or "no"
        WHILE repeatProgram is NOT "yes" AND repeatProgram is NOT "no"
            DISPLAY "Invalid choice. Enter yes or no."
            PROMPT "Do you want to calculate another month? (yes/no): "
            GET repeatProgram
        END WHILE

    END WHILE

    // Final message when the user chooses not to repeat the program
    DISPLAY "Thank you for using the Monthly Budget Calculator."

END
